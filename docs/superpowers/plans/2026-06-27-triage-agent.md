# Triage Issue Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add two stateless agents that auto-classify new GitHub issues on creation and run an interactive triage state machine when a maintainer posts `/triage`.

**Architecture:** `issue-classifier` dispatches on `issues.opened` to label category + priority + `needs-triage`; `issue-triager` dispatches on `/triage` issue comments to run the full state machine (adjust labels, set difficulty, post agent brief or needs-info, close wontfix). Both are stateless — GitHub labels are the source of truth. Follows the same `defineAgent` + gateway + tools pattern as `pr-reviewer`.

**Tech Stack:** `@flue/runtime` (defineAgent, defineTool), `@octokit/rest`, `valibot`, `bun:test`

## Global Constraints

- Use `bun test` not jest/vitest
- Use Bun imports (`bun:test`, `bun:sqlite`, etc.) not Node equivalents
- Model: `deepseek/deepseek-v4-flash`; classifier has no thinking, triager uses `thinkingLevel: 'medium'`
- All GitHub calls go through the gateway — no Octokit calls in agent files
- Every triage comment must begin with `> *This was generated by AI during triage.*`
- Bot-authored issues are skipped (reuse `REVIEW_BOT_PRS` env guard)
- Gateway tests mock `@octokit/rest` with `mock.module` (same pattern as `pull-requests.test.ts`)

import { Octokit } from '@octokit/rest';

const client = new Octokit({ auth: process.env.GITHUB_TOKEN });

export type IssueRef = { owner: string; repo: string; issueNumber: number };
export type LabelDef = { name: string; color: string; description?: string };

export type IssueComment = {
	id: number;
	author: string;
	body: string;
	createdAt: string;
};

export type Issue = {
	number: number;
	title: string;
	body: string;
	state: 'open' | 'closed';
	labels: string[];
	comments: IssueComment[];
	author: string;
};

export interface IssueGateway {
	get(): Promise<Issue>;
	addLabels(names: string[]): Promise<void>;
	removeLabel(name: string): Promise<void>;
	postComment(body: string): Promise<void>;
	close(): Promise<void>;
	ensureLabels(defs: LabelDef[]): Promise<void>;
	readRepoFile(path: string): Promise<string | null>;
}

export const MISSING_LABELS: LabelDef[] = [
	{ name: 'needs-triage',    color: 'e4e669', description: 'Awaiting triage' },
	{ name: 'needs-info',      color: 'd876e3', description: 'Waiting on reporter' },
	{ name: 'ready-for-agent', color: '0075ca', description: 'Fully specified, agent-ready' },
	{ name: 'ready-for-human', color: '008672', description: 'Requires human implementation' },
	{ name: 'difficulty:hard', color: 'EF4444', description: 'Significant effort required' },
];

export function createIssueGateway(ref: IssueRef): IssueGateway {
	const { owner, repo, issueNumber: issue_number } = ref;

	return {
		async get() {
			const [issue, comments] = await Promise.all([
				client.rest.issues.get({ owner, repo, issue_number }),
				client.paginate(client.rest.issues.listComments, { owner, repo, issue_number, per_page: 100 }),
			]);
			return {
				number: issue.data.number,
				title: issue.data.title,
				body: issue.data.body ?? '',
				state: issue.data.state as 'open' | 'closed',
				labels: issue.data.labels.map((l) => (typeof l === 'string' ? l : (l.name ?? ''))),
				comments: comments.map((c) => ({
					id: c.id,
					author: c.user?.login ?? '',
					body: c.body ?? '',
					createdAt: c.created_at,
				})),
				author: issue.data.user?.login ?? '',
			};
		},

		async addLabels(names) {
			await client.rest.issues.addLabels({ owner, repo, issue_number, labels: names });
		},

		async removeLabel(name) {
			try {
				await client.rest.issues.removeLabel({ owner, repo, issue_number, name });
			} catch (err) {
				if (err && typeof err === 'object' && 'status' in err && err.status === 404) return;
				throw err;
			}
		},

		async postComment(body) {
			await client.rest.issues.createComment({ owner, repo, issue_number, body });
		},

		async close() {
			await client.rest.issues.update({ owner, repo, issue_number, state: 'closed' });
		},

		async ensureLabels(defs) {
			const existing = await client.paginate(client.rest.issues.listLabelsForRepo, { owner, repo, per_page: 100 });
			const existingNames = new Set(existing.map((l) => l.name));
			await Promise.all(
				defs
					.filter((d) => !existingNames.has(d.name))
					.map((d) =>
						client.rest.issues.createLabel({
							owner,
							repo,
							name: d.name,
							color: d.color,
							description: d.description ?? '',
						}),
					),
			);
		},

		async readRepoFile(path) {
			try {
				const res = await client.rest.repos.getContent({ owner, repo, path });
				const data = res.data;
				if (Array.isArray(data) || data.type !== 'file' || typeof data.content !== 'string') return null;
				return Buffer.from(data.content, 'base64').toString('utf8');
			} catch (err) {
				if (err && typeof err === 'object' && 'status' in err && err.status === 404) return null;
				throw err;
			}
		},
	};
}
```

import { defineTool } from '@flue/runtime';
import * as v from 'valibot';
import type { IssueGateway } from './issues.ts';
import { MISSING_LABELS } from './issues.ts';

export function createClassifierTools(gw: IssueGateway) {
	return [
		defineTool({
			name: 'ensure_triage_labels',
			description:
				'Create missing triage labels in this repo (needs-triage, needs-info, ready-for-agent, ready-for-human, difficulty:hard). Call once before adding any labels.',
			run: () => gw.ensureLabels(MISSING_LABELS),
		}),

		defineTool({
			name: 'add_labels',
			description: 'Apply one or more labels to this issue.',
			input: v.object({
				names: v.array(v.pipe(v.string(), v.minLength(1)), 'Label names to apply.'),
			}),
			run: ({ input }) => gw.addLabels(input.names),
		}),

		defineTool({
			name: 'post_comment',
			description: 'Post a comment on this issue.',
			input: v.object({
				body: v.pipe(v.string(), v.minLength(1), v.description('Comment body in Markdown.')),
			}),
			run: ({ input }) => gw.postComment(input.body),
		}),
	];
}

export function createTriagerTools(gw: IssueGateway) {
	return [
		defineTool({
			name: 'get_issue',
			description:
				'Get the issue title, body, current labels, all comments (id, author, body, createdAt), and the issue author.',
			run: () => gw.get(),
		}),

		defineTool({
			name: 'ensure_triage_labels',
			description:
				'Create missing triage labels (needs-triage, needs-info, ready-for-agent, ready-for-human, difficulty:hard). Call once at the start of triage.',
			run: () => gw.ensureLabels(MISSING_LABELS),
		}),

		defineTool({
			name: 'add_labels',
			description: 'Apply one or more labels to this issue.',
			input: v.object({
				names: v.array(v.pipe(v.string(), v.minLength(1)), 'Label names to apply.'),
			}),
			run: ({ input }) => gw.addLabels(input.names),
		}),

		defineTool({
			name: 'remove_label',
			description: 'Remove a label from this issue. Safe to call even if the label is not present.',
			input: v.object({
				name: v.pipe(v.string(), v.minLength(1), v.description('Label name to remove.')),
			}),
			run: ({ input }) => gw.removeLabel(input.name),
		}),

		defineTool({
			name: 'post_comment',
			description: 'Post a comment on this issue.',
			input: v.object({
				body: v.pipe(v.string(), v.minLength(1), v.description('Comment body in Markdown.')),
			}),
			run: ({ input }) => gw.postComment(input.body),
		}),

		defineTool({
			name: 'close_issue',
			description: 'Close this issue. Use only for wontfix outcomes.',
			run: () => gw.close(),
		}),

		defineTool({
			name: 'read_repo_file',
			description:
				'Read a file from the default branch of the repository by its path. Returns text content or null if missing or binary. Use to read CONTEXT.md, docs/adr/**, or any file needed to check for existing implementations.',
			input: v.object({
				path: v.pipe(v.string(), v.minLength(1), v.description('Repo-relative file path.')),
			}),
			run: ({ input }) => gw.readRepoFile(input.path),
		}),
	];
}

description: State machine for triaging GitHub issues — classify, verify, and write durable agent briefs. Use when running /triage on an issue.

# Issue Triage

Move this issue through the triage state machine. Every issue should carry exactly one category label (`bug` or `enhancement`) and exactly one state label.

Every comment you post **must** start with:

```
> *This was generated by AI during triage.*
```

followed by a blank line before the content.

## Labels

**Category:** `bug` · `enhancement`

**State:** `needs-triage` → `needs-info` | `ready-for-agent` | `ready-for-human` | `wontfix`

**Priority:** `priority:critical` · `priority:high` · `priority:medium` · `priority:low`

**Difficulty:** `difficulty:easy` · `difficulty:medium` · `difficulty:hard`

## Full evaluation (no directive)

1. Call `get_issue` to read the title, body, current labels, and all comments.
2. Read prior triage notes from comments so you don't re-ask resolved questions.
3. Adjust the category label if the auto-classifier was wrong (remove the old one, add the correct one).
4. Adjust the priority label if needed (remove old, add correct).
5. Check the codebase: call `read_repo_file` with `CONTEXT.md` and relevant ADR files in `docs/adr/` to check whether this behavior is already implemented or was previously rejected.
6. Determine the state and (if applicable) difficulty. Act per the outcome table below.

## Outcome table

| State | Add labels | Post comment | Close? |
|---|---|---|---|
| `ready-for-agent` | state + `difficulty:*` | Agent brief (see template) | No |
| `ready-for-human` | state + `difficulty:*` | Brief + reason not delegatable | No |
| `needs-info` | state | Needs-info comment (see template) | No |
| `wontfix` | — | Explanation then close | Yes |

For `ready-for-agent` and `ready-for-human`: remove the `needs-triage` or `needs-info` label first, then add the new state label.

## Directive shortcut

If a directive was given (e.g. `/triage wontfix`), apply it directly without full evaluation:
1. Remove the current state label.
2. Add the directive state label.
3. If `ready-for-agent`: still write the agent brief.
4. If `wontfix`: close the issue after posting the explanation.
5. Post a brief confirmation.

## Agent brief template

```markdown
> *This was generated by AI during triage.*

## Agent Brief

**Category:** bug / enhancement
**Summary:** one-line description of what needs to happen

**Current behavior:**
Describe what happens now. For bugs: the broken behavior. For enhancements: the status quo.

**Desired behavior:**
Describe the end state. Be specific about edge cases and error conditions.

**Key interfaces:**
- `TypeName` — what needs to change and why
- `functionName()` — current vs desired return type or behavior

**Acceptance criteria:**
- [ ] Specific, testable criterion
- [ ] Another criterion

**Out of scope:**
- Thing that should NOT be changed
```

## Needs-info template

```markdown
> *This was generated by AI during triage.*

## Triage Notes

**What we've established so far:**
- point 1

**What we still need from you (@reporter):**
- Specific question 1 (not "please provide more info")
- Specific question 2
```

## Calibration

- `priority:critical` — data loss, security breach, system completely down
- `priority:high` — significant feature broken, blocking multiple users, no workaround
- `priority:medium` — notable problem with a workaround available
- `priority:low` — minor cosmetic or convenience issue

- `difficulty:easy` — isolated change, well-understood area, < 2 hours
- `difficulty:medium` — requires understanding multiple components, 2–8 hours
- `difficulty:hard` — significant effort, cross-cutting concerns, architecture decisions, > 8 hours
```

Your input is a JSON object with:

// --- Issue or PR comment ----------------------------------------------------
if (delivery.name === 'issue_comment' && delivery.payload.action === 'created') {
    const { repository, issue, comment, sender } = delivery.payload;
    if (isSelf(sender)) return;

    if (!issue.pull_request) {
        // Issue comment — /triage command
        const body = comment.body.trim();
        if (!body.toLowerCase().startsWith('/triage')) return;
        const rest = body.slice('/triage'.length).trim().toLowerCase();
        const directive = rest || undefined;
        const validDirectives = ['needs-triage', 'needs-info', 'ready-for-agent', 'ready-for-human', 'wontfix'];
        if (directive && !validDirectives.includes(directive)) return;
        const owner = repository.owner.login;
        const repo = repository.name;
        await dispatch(issueTriager, {
            id: channel.conversationKey({ owner, repo, issueNumber: issue.number }),
            input: {
                command: 'triage',
                owner,
                repo,
                issue: issue.number,
                directive,
                requester: sender.login,
            },
        });
        return;
    }

    // PR comment — review commands
    const normalized = comment.body.trim().toLowerCase().replace(/^\//, '').trim();
    const command =
        normalized === 'full review' ? 'full' : normalized === 'review' ? 'incremental' : null;
    if (!command) return;

    const owner = repository.owner.login;
    const repo = repository.name;
    await dispatch(prReviewer, {
        id: channel.conversationKey({ owner, repo, issueNumber: issue.number }),
        input: {
            type: 'github.pull_request.review',
            command,
            deliveryId: delivery.deliveryId,
            owner,
            repo,
            pr: issue.number,
            reason: `"${normalized}" command from @${sender.login}.`,
        },
    });
    return;
}
